set(UNIX_DIR ${CMAKE_SOURCE_DIR}/src/unix)
set(QT_DIR   ${CMAKE_SOURCE_DIR}/src/qt)

# Platform helpers. The plain-C platform layer of the SDL frontend is reused
# as is: it needs SDL only for timers and the settings directory, and SDL is
# already a dependency for joysticks.
add_library(plat OBJECT
    haiku_paths.cpp
    haiku_platform.cpp
    ${UNIX_DIR}/sdl_plat.c
    ${UNIX_DIR}/sdl_plat_unix.c
    ${UNIX_DIR}/unix_netsocket.c
    ${UNIX_DIR}/unix_dynld.c
    ${UNIX_DIR}/dummy_cdrom_ioctl.c
    ${UNIX_DIR}/dummy_floppy_ioctl.c
    ${QT_DIR}/sdl_joystick.c
)
target_include_directories(plat PRIVATE ${UNIX_DIR})
# Use haiku_paths.cpp instead of the SDL UI's folder functions.
target_compile_definitions(plat PRIVATE HAIKU_NATIVE_PATHS)
target_compile_definitions(plat PUBLIC _FILE_OFFSET_BITS=64)

if (NOT CPPTHREADS)
    target_sources(plat PRIVATE ${UNIX_DIR}/unix_thread.c)
endif()

# Status-bar icons, shared with the Qt UI, are compiled into the binary so it
# needs no files next to it; haiku_machinestatus.cpp decodes them.
set(HAIKU_ICONS
    # status bar
    floppy_35 floppy_525 floppy_disabled cdrom cdrom_disabled cdrom_mute
    hard_disk network sound cassette cartridge rdisk rdisk_disabled zip jaz
    syquest mo mo_disabled tape tape_disabled usb_port recompiler interpreter
    active write_active write_protected
    num_lock_on num_lock_off caps_lock_on caps_lock_off scroll_lock_on
    scroll_lock_off kana_lock_on kana_lock_off
    # toolbar and menus
    pause run fast_forward hard_reset acpi_shutdown send_cad send_cae
    settings take_screenshot copy_screenshot take_raw_screenshot
    copy_raw_screenshot eject browse new
    # VM manager
    86Box-gray new_vm
)
set(HAIKU_ICON_SOURCE ${CMAKE_CURRENT_BINARY_DIR}/haiku_icons.cpp)
set(HAIKU_ICON_CODE "/* Generated by src/haiku/CMakeLists.txt. */\n#include \"haiku_icons.hpp\"\n\n")
set(HAIKU_ICON_TABLE "")
foreach(icon IN LISTS HAIKU_ICONS)
    set(icon_file ${QT_DIR}/icons/${icon}.ico)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${icon_file})
    file(READ ${icon_file} icon_hex HEX)
    string(LENGTH "${icon_hex}" icon_len)
    math(EXPR icon_size "${icon_len} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," icon_bytes "${icon_hex}")
    string(MAKE_C_IDENTIFIER "icon_${icon}" icon_var)
    string(APPEND HAIKU_ICON_CODE "static const uint8_t ${icon_var}[] = { ${icon_bytes} };\n")
    string(APPEND HAIKU_ICON_TABLE "    { \"${icon}\", ${icon_var}, ${icon_size} },\n")
endforeach()
string(APPEND HAIKU_ICON_CODE "\nconst haiku_icon_t haiku_icons[] = {\n${HAIKU_ICON_TABLE}    { nullptr, nullptr, 0 }\n};\n")
file(CONFIGURE OUTPUT ${HAIKU_ICON_SOURCE} CONTENT "${HAIKU_ICON_CODE}" @ONLY)

# in the image history, as Qt does) instead of the one in sdl_cdrom.c.
target_compile_definitions(ui PRIVATE HAIKU_NATIVE_MEDIA)
# The memory viewer's disassembler, Zydis, from the Qt UI's copy (as in Qt).
if(DISASM)
    add_subdirectory(${QT_DIR}/zydis ${CMAKE_CURRENT_BINARY_DIR}/zydis)
    target_compile_definitions(ui PRIVATE USE_DISASM)
    target_link_libraries(ui PRIVATE zydis)
endif()

# The MIDI device choices of the device configuration dialog (as in Qt).
if(RTMIDI)
    target_compile_definitions(ui PRIVATE USE_RTMIDI)
endif()

target_link_libraries(plat PRIVATE SDL3::Headers)
# be: Interface/App Kits, tracker: BFilePanel, translation: VM manager
# screenshots and icons, GL: the OpenGL renderer (and BGLView), libgame: set_mouse_position().
# libgame is passed as -lgame because 86Box has its own target named "game"
# (src/game, the gameport code), which a plain "game" would resolve to.
target_link_libraries(86Box SDL3::SDL3 be tracker translation GL -lgame ${CMAKE_DL_LIBS})
